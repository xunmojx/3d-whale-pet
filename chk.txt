let startPos = { x: undefined, y: undefined }
  try {
    // 起始位置：屏幕左下角（贴任务栏上方一点，留 24 的边距）。
    // workArea 是物理像素、窗口坐标是 DIP，这台屏 150% 缩放，必须换算。
    const disp = screen.getPrimaryDisplay()
    const sf = disp.scaleFactor || 1
    const waX = disp.workArea.x / sf
    const waY = disp.workArea.y / sf
    const waW = disp.workArea.width / sf
    const waH = disp.workArea.height / sf
    const margin = 24
    startPos = {
      x: Math.round(waX + margin),
      y: Math.round(waY + waH - winH() - margin),
    }
  } catch { /* 拿不到屏幕就交给 Electron 决定 */ }
