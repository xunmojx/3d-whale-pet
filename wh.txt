ipcMain.on('pet:clear-local-data', async () => {
  const result = { ok: false }
  wiping = true
  try {
    // 1) 配置文件（几 KB，能直接删）
    const cfgPaths = new Set([CONFIG_PATH, join(app.getPath('userData'), 'whale-pet-config.json')].filter(Boolean))
    for (const p of cfgPaths) {
      try { if (fs.existsSync(p)) fs.unlinkSync(p) } catch { /* 删不掉也继续 */ }
    }

    // 2) 桌宠本体（180MB）：现在正被自己这个进程占用着，删不动。
    //    交给一个独立进程，等 3 秒（本进程退出后）再删。
    const rt = join(process.env.APPDATA || join(require('node:os').homedir(), 'AppData', 'Roaming'),
                    'whale-pet-app', 'runtime')
    if (fs.existsSync(rt)) {
      try {
        const { spawn } = require('node:child_process')
        const child = spawn('cmd', ['/c', `timeout /t 3 /nobreak >nul & rmdir /s /q "${rt}"`], {
          detached: true,
          stdio: 'ignore',
          windowsHide: true,
        })
        child.unref()
      } catch { /* 起不来就算了 */ }
    }
    result.ok = true
  } catch (err) {
    result.error = err && err.message ? err.message : String(err)
    wiping = false
  }
  sendToWin('pet:clear-local-data-result', result)
  if (result.ok) {
    setTimeout(() => { try { app.quit() } catch { /* 忽略 */ } }, 800)
  }
})
